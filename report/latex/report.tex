% Float placement. LaTeX's defaults (topfraction 0.7, totalnumber 3, textfraction 0.2)
% are too conservative for a chapter with this many tables: floats queue up and get
% flushed to the end of the chapter, pages away from the text that references them.
% Loosening them keeps each float on or beside its discussion, which is what a reader
% needs when the text says "see table N".
%
% floatpagefraction is the one that decides whether a page may hold floats and
% NOTHING else. At the default 0.5 -- and still at 0.75 -- two adjacent floats fill
% enough of a page for LaTeX to give them one of their own, which is how tables 3.1,
% 3.2, 3.7, 3.8 and figure 3.6 each ended up on a text-free page. At 0.95 LaTeX prefers
% a top float with text under it, and only a float that fits nowhere else takes a page.
% Float placement is global: re-check the WHOLE pdf after touching a float or a caption.
\renewcommand{\topfraction}{0.92}
\renewcommand{\bottomfraction}{0.8}
\renewcommand{\textfraction}{0.07}
\renewcommand{\floatpagefraction}{0.95}
\setcounter{topnumber}{3}
\setcounter{bottomnumber}{2}
\setcounter{totalnumber}{5}
